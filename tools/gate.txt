# The gate: every check the tree is held to without a compiler of its own, one row a check, in the
# order they run. The ONE list: `just lint` and blaze.yaml's `scripts` job both run it through
# scripts/src/bin/gate.rs (`{"command": "run"}`), which first checks that neither runs a check
# outside it (`{"command": "check"}`: a gate that runs different things in two places is two
# gates). The cargo job (check and the workspace's tests) and the lean job are not here: they
# build.
#
# A row:  name | seconds | base | command
#   name     unique, what the run prints
#   seconds  its timeout (rule 69): an expiry is its failure, named, never retried
#   base     `base`: the command's last word is a JSON object in single quotes, and a run given a
#            base (CI's pull request, merge queue or push) gets `"base": "<sha>"` added to it, so
#            a ratchet is held to the base's rows too; `-`: it takes none
#   command  bash, run from the checkout's root; `$BLAZE_SCRIPTS` is the built scripts' directory
#            (the run's `scripts` field)

# The repository's Python workspace (pyproject.toml, uv.lock) lints. Some scripts
# ask rustc for the host when imported: in the workspace's Python, with its helpers.
python lint | 300 | - | uv run --frozen ruff check
python imports | 300 | - | uv run --frozen python harness/imports.py
# The list of tests that may fail holds together and is as its recorder writes it.
failures selftest | 120 | - | uv run --frozen python harness/failures.py selftest
# A row or a test run counts only with proof that it ran (harness/v2/ran.py).
ran selftest | 120 | - | uv run --frozen python harness/v2/ran.py selftest
# library/DIGEST is the digest of library/ as it is (a stale one stops every fresh home).
library digest | 120 | - | uv run --frozen python harness/library.py check '{}'
# The agent rules: tools/agents/rules.ts type-checks (under bun: tsc's script asks for node), and
# AGENTS.md and the rules in .agents/skills/*/SKILL.md are what it generates. The site's lock is
# installed as it is (`--frozen-lockfile` refuses a stale one).
site packages | 300 | - | bun install --frozen-lockfile --cwd site
agent rules types | 300 | - | bun site/node_modules/.bin/tsc -p tools/agents
agent rules | 120 | - | bun tools/agents/rules.ts --check
# Each doc says what it is, the current ones hold against the tree, docs/README.md is their index,
# and no Python file is added (tools/python-allowed.txt only shrinks; scripts/src/bin/docs.rs).
docs selftest | 120 | - | "$BLAZE_SCRIPTS/docs" selftest
docs | 300 | - | "$BLAZE_SCRIPTS/docs" check
# No new place in blaze's code starts rustc, rustdoc or cargo (docs/no-rustc.md).
rustc sites | 300 | - | "$BLAZE_SCRIPTS/rustcsites"
# The objectives ledger's rows are well formed and its table is current.
ledger selftest | 120 | - | "$BLAZE_SCRIPTS/ledger" selftest
ledger | 120 | - | "$BLAZE_SCRIPTS/ledger" '{"command": "check"}'
# Which packages a change touches.
changed selftest | 120 | - | "$BLAZE_SCRIPTS/changed" selftest
# Every stated invariant has a control beside it, and docs/v2/plan/invariants.md lists them.
invariants selftest | 120 | - | "$BLAZE_SCRIPTS/invariants" selftest
invariants | 300 | - | "$BLAZE_SCRIPTS/invariants" '{"command": "check"}'
# The Lean proofs as far as no Lean is needed: lean/theorems.json names statements that are

# Each hash domain in crates/ is written once outside tests and is under a cache of

hash domains | 300 | - | "$BLAZE_SCRIPTS/domains" '{"root": "."}'
# The ratchets: each crate's count equals its row, a crate with no row is held to 0, no row is
# above the base's, and a crate over its row has its new sites named. The global heap
# (alloc-sites.txt), untyped errors (untyped-errors.txt), third-party packages (third-party.txt,
# and reached.txt: the crates the binaries reach), std and alloc lines (std-alloc.txt).
global heap selftest | 120 | - | "$BLAZE_SCRIPTS/allocs" selftest
global heap | 300 | base | "$BLAZE_SCRIPTS/allocs" '{"command": "check"}'
untyped errors selftest | 120 | - | "$BLAZE_SCRIPTS/untyped" selftest
untyped errors | 300 | base | "$BLAZE_SCRIPTS/untyped" '{"command": "check"}'
third party selftest | 120 | - | "$BLAZE_SCRIPTS/thirdparty" selftest
third party (and reached) | 300 | base | "$BLAZE_SCRIPTS/thirdparty" '{"command": "check"}'
# Every member names each dependency through the root's [workspace.dependencies] (`workspace =
# true`, no path or version of its own), and the root names every member there once by its path.
manifests | 120 | - | "$BLAZE_SCRIPTS/thirdparty" '{"command": "manifests-check"}'
# D180: the third-party packages cargo's resolve reaches, target 0 (printed each run), held at
# tools/ratchets/thirdparty.txt until it is empty, then refused outright. `--offline` reads the
# local registry cache and never the network; `--locked` refuses a Cargo.lock the manifests have
# outgrown, naming it, and nothing updates it here.
third party reach | 300 | - | m=$(mktemp) && trap 'rm -f "$m"' EXIT && { timeout 240 cargo fetch --locked --quiet && timeout 240 cargo metadata --format-version 1 --offline --locked > "$m" || { echo 'third party reach: `cargo metadata --offline --locked` refused (its error above): a stale Cargo.lock is refused, not updated (update it with the network and commit it)' >&2; exit 1; }; } && "$BLAZE_SCRIPTS/thirdparty" "{\"command\": \"reach-check\", \"metadata\": \"$m\"}"
# D180: blaze-sys references no C-library symbol on Linux and only the libSystem symbols of

sys imports selftest | 120 | - | "$BLAZE_SCRIPTS/sys_imports" selftest
ambient selftest | 120 | - | "$BLAZE_SCRIPTS/ambient" selftest
